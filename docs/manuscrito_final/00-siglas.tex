% Lista de abreviaturas e siglas: termos utilizados nos capítulos.
\pretextualchapter{Lista de abreviaturas e siglas}
\begin{description}[labelsep=5mm,labelwidth=22mm,leftmargin=27mm,align=left]
  \item[AMI] Advanced Metering Infrastructure (infraestrutura de medição avançada)
  \item[CIS] Customer Information System (sistema de informações comerciais)
  \item[CLI] Command Line Interface (interface de linha de comando)
  \item[COPOD] Copula-Based Outlier Detection (detecção de anomalias baseada em cópulas)
  \item[CPD] Change Point Detection (detecção de mudanças de regime)
  \item[CSV] Comma-Separated Values (valores separados por vírgulas)
  \item[CUSUM] Cumulative Sum (soma cumulativa)
  \item[CV] Coeficiente de variação
  \item[DTW] Dynamic Time Warping (alinhamento temporal dinâmico)
  \item[ECOD] Empirical Cumulative Distribution Functions for Outlier Detection (detecção de anomalias por distribuições acumuladas empíricas)
  \item[EDA] Exploratory Data Analysis (análise exploratória de dados)
  \item[EMV] Expected Monetary Value (valor monetário esperado)
  \item[ETL] Extract, Transform, Load (extração, transformação e carga)
  \item[FA] Energia ativa no sentido direto (consumo)
  \item[GD] Geração distribuída
  \item[GEO] Fonte de dados geográficos e topológicos da rede
  \item[HBOS] Histogram-Based Outlier Score (escore de anomalia baseado em histogramas)
  \item[IF] Isolation Forest
  \item[JSON] JavaScript Object Notation (notação de objetos JavaScript)
  \item[LOF] Local Outlier Factor (fator de anomalia local)
  \item[MAD] Median Absolute Deviation (desvio absoluto mediano)
  \item[MDM] Meter Data Management (gerenciamento de dados de medição)
  \item[ML] Machine Learning (aprendizado de máquina)
  \item[MoE] Mixture of Experts (mistura de especialistas)
  \item[NIO] Identificador do medidor físico no conjunto de dados
  \item[OOT] Out-of-Time (avaliação fora do período de ajuste)
  \item[ORCA] Sistema de origem de dados de medição (MDM)
  \item[OS] Ordem de serviço
  \item[PCA] Principal Component Analysis (análise de componentes principais)
  \item[PNT] Perdas não técnicas
  \item[PRODIST] Procedimentos de Distribuição de Energia Elétrica no Sistema Elétrico Nacional
  \item[RA] Energia ativa no sentido reverso (injeção)
  \item[RAM] Random Access Memory (memória de acesso aleatório)
  \item[SQL] Structured Query Language (linguagem de consulta estruturada)
  \item[TCC] Trabalho de Conclusão de Curso
  \item[TOI] Termo de ocorrência e inspeção
  \item[TUI] Terminal User Interface (interface de usuário em terminal)
  \item[UC] Unidade consumidora
  \item[UFPR] Universidade Federal do Paraná
\end{description}
